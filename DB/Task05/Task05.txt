CREATE TABLE PATIENT(
PATIENT_ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY NOT NULL,
PATIENT_NAME VARCHAR(255) ,
PATIENT_AGE NUMBER);

CREATE TABLE DOCTORS_AND_PATIENTS (
DOCTOR_ID NUMBER ,
PATIENT_ID NUMBER ,
CONSTRAINT DOCTOR_ID_FK FOREIGN KEY (DOCTOR_ID) REFERENCES DOCTOR (DOCTOR_ID) ,
CONSTRAINT PATIENT_ID_FK FOREIGN KEY (PATIENT_ID) REFERENCES PATIENT (PATIENT_ID) ,
CONSTRAINT PATIENT_DOCTOR UNIQUE  (DOCTOR_ID,PATIENT_ID)
);


SELECT 
    d.NAME,
    p.PATIENT_NAME
FROM DOCTOR d
LEFT JOIN DOCTORS_AND_PATIENTS dp 
    ON d.DOCTOR_ID = dp.DOCTOR_ID
LEFT JOIN PATIENT p 
    ON p.PATIENT_ID = dp.PATIENT_ID;

    d.NAME AS DOCTOR_NAME,
