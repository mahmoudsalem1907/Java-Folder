SELECT AVG(SALARY) AVG FROM EMPLOYEES e ;
SELECT COUNT(*) COUNTE FROM EMPLOYEES e ;
SELECT MAX(SALARY) MAX FROM EMPLOYEES e ;
SELECT MIN(SALARY) MIN FROM EMPLOYEES e ;
SELECT SUM(SALARY) SUM FROM EMPLOYEES e ;
SELECT e.DEPARTMENT_ID ,AVG(SALARY) AVG FROM EMPLOYEES e GROUP BY e.DEPARTMENT_ID ;
SELECT e.JOB_ID  ,COUNT(*) COUNT FROM EMPLOYEES e GROUP BY e.JOB_ID ;
SELECT e.DEPARTMENT_ID ,SUM(SALARY) SUM FROM EMPLOYEES e GROUP BY e.DEPARTMENT_ID HAVING SUM(e.SALARY) >50000 ;
SELECT AVG(e.COMMISSION_PCT) AVG FROM EMPLOYEES e WHERE e.COMMISSION_PCT IS NOT NULL  ;
SELECT COUNT(*) FROM EMPLOYEES e WHERE e.SALARY > 10000;
SELECT MAX(SALARY) MAX ,e.JOB_ID ,MIN(SALARY) MIN FROM EMPLOYEES e GROUP BY e.JOB_ID ;
SELECT SUM(SALARY) SUM FROM EMPLOYEES e WHERE e.MANAGER_ID IS NOT NULL ;
SELECT e.DEPARTMENT_ID , e.JOB_ID , SUM(SALARY) SUM FROM EMPLOYEES e GROUP BY e.DEPARTMENT_ID , JOB_ID  ;
SELECT e.JOB_ID , COUNT(*) COUNT FROM EMPLOYEES e GROUP BY e.JOB_ID HAVING COUNT(*) > 5;
SELECT COUNT(*) COUNT ,MIN(SALARY) MIN,MAX(SALARY) MAX, AVG(SALARY) AVG FROM EMPLOYEES e GROUP BY e.DEPARTMENT_ID;
SELECT e.DEPARTMENT_ID  FROM EMPLOYEES e GROUP BY e.DEPARTMENT_ID HAVING AVG(SALARY) >8000 AND COUNT(*) > 10    ;
SELECT e.DEPARTMENT_ID,SUM(e.SALARY) AS TOTAL_SALARY FROM EMPLOYEES e GROUP BY e.DEPARTMENT_ID ORDER BY TOTAL_SALARY DESC FETCH FIRST 1 ROW ONLY;
SELECT SUM(SALARY) SUM ,e.DEPARTMENT_ID  ,AVG(SALARY) AVG FROM EMPLOYEES e GROUP BY e.DEPARTMENT_ID  ;
